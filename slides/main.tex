\begin{frame}{Evaluation Metrics}
  \par\vspace{0.14cm}
  {\small\color{uetdark}
  \renewcommand{\arraystretch}{1.55}
  \begin{tabularx}{\textwidth}{@{}>{\raggedright\arraybackslash}p{1.65cm} *{4}{>{\raggedright\arraybackslash}X}@{}}
    \toprule
    \rowcolor{uetaccentbg}
    \key{Metric} & \key{PSNR (dB)} & \key{SSIM} & \key{FPS} & \key{QSS}\\
    \midrule
    \textbf{Full name} & Peak signal-to-noise ratio & Structural similarity index & Frames per second & Quality-Speed Score\\
    \midrule
    \textbf{What it measures} & Pixel accuracy against a clean reference. & Structure, contrast, and luminance similarity. & Number of images processed per second. & Balance between quality and speed.\newline $\mathrm{QSS}=\dfrac{\mathrm{FPS}}{\mathrm{MSE}}$\\
    \midrule
    \textbf{How to interpret it} & Higher values indicate lower pixel error. & Values closer to 1 indicate greater similarity. & Higher values indicate faster processing. & Higher values favor faster processing and lower pixel error.\\
    \bottomrule
  \end{tabularx}}\par
  \vspace{0.25cm}
  {\scriptsize\color{uetgray}MSE: mean squared error against a clean reference. Use the same image value range for MSE.\par
  Compare FPS and QSS using the same hardware, image resolution, and batch size.}
  % Metric definitions: scikit-image metrics documentation (PSNR and SSIM).
  % https://scikit-image.org/docs/stable/api/skimage.metrics.html
  % FPS is throughput; the measurement setup must be held constant.

\section{Next Steps}
% 13. Planned architecture experiment and corresponding hyperparameter tuning.
\begin{frame}{Next Steps}
  \begin{block}{1. Experiment with Mamba}
    Replace Attention with Mamba in the low-frequency recovery block (LFRN)
    and compare it with the current Attention-based baseline.
  \end{block}
  \vspace{0.15cm}
  \begin{block}{2. Optimize the corresponding hyperparameters}
    Tune Mamba state size, expansion factor, and block depth, together with
    training settings such as learning rate, batch size, and weight decay.
  \end{block}
  \vspace{0.15cm}
  \begin{block}{3. Evaluate the quality--speed balance}
    Compare PSNR, SSIM, FPS, and QSS on the same dataset split and hardware;
    select the configuration that best meets the target metrics.
  \end{block}
\end{frame}
